# Create the directory for the share
mkdir -p /home/haas/Haas_Data_collect/machines/st30

sudo cp st30.service /etc/systemd/system/st30.service
sudo systemctl daemon-reload
sudo systemctl enable st30.service
sudo systemctl start st30.service
sudo systemctl status st30.service


# Create the share configuration in /etc/samba/smb.conf
# use sudo micro /etc/samba/smb.conf
or
# sudo nano /etc/samba/smb.conf

[st30]
    comment = Logger for ST30
    path = /home/haas/Haas_Data_collect/machines/st30
    read only = no
    browsable = yes
    writable = yes
    public = no
    valid users = @HaasGroup, haas # Ensure the user is valid
    force user = haas
    force group = HaasGroup
    create mask = 0664
    force create mode = 0664
    directory mask = 0775
    force directory mode = 0775
