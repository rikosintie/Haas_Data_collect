sudo cp haas-st40.service /etc/systemd/system/haas-st40.service
sudo systemctl daemon-reload
sudo systemctl enable haas-st40.service
sudo systemctl start haas-st40.service
sudo systemctl status haas-st40.service

# Create the directory for the share

mkdir /home/haas/st40

# Create the share configuration

[st40]
    comment = Logger for ST40
    path = /home/haas/Haas_Data_collect/machines/st40
    read only = no
    browsable = yes
    writable = yes
    public = no
    valid users = @HaasGroup, haas # Ensure the user is valid
    force user = haas
    force group = HaasGroup
    create mask = 0664
    force create mode = 0664
    directory mask = 0775
    force directory mode = 0775
