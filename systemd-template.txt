# Create the directory for the share
mkdir -p /home/haas/Haas_Data_collect/machines/{{name}}

sudo cp {{name}}.service /etc/systemd/system/{{name}}.service
sudo systemctl daemon-reload
sudo systemctl enable {{name}}.service
sudo systemctl start {{name}}.service
sudo systemctl status {{name}}.service


# Create the share configuration in /etc/samba/smb.conf
# use sudo micro /etc/samba/smb.conf
or
# sudo nano /etc/samba/smb.conf

[{{name}}]
    comment = {{description}}
    path = /home/haas/Haas_Data_collect/machines/{{name}}
    read only = no
    browsable = yes
    writable = yes
    public = no
    valid users = @HaasGroup, haas # Ensure the user is valid
    force user = haas
    force group = HaasGroup
    create mask = 0664
    force create mode = 0664
    directory mask = 0775
    force directory mode = 0775
